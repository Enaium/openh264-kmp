#
# Compiles OpenH264 (common + decoder) and the KMP C++ shim into one static
# archive and, optionally, the JNI shared library or the standalone
# WebAssembly module.
#
#   cmake -S jni -B build -DBUILD_JNI=ON               desktop JVM / Android JNI
#   cmake -S jni -B build                              static archive only
#   emcmake cmake -S jni -B build-wasm -DBUILD_WASM=ON WebAssembly
#
# OpenH264 ships hand-written Makefiles, not a CMake project, so this file
# lists the sources from codec/common/targets.mk and codec/decoder/targets.mk.
# Assembly is disabled on every target: the binding runs the portable C/C++
# kernels, which keeps one code path for all platforms and removes NASM and
# per-architecture assembler requirements from the build.
#
# OPENH264_DIR points at the OpenH264 checkout to compile; the default is the
# git submodule next to this directory.
#
cmake_minimum_required(VERSION 3.20)

project(openh264_kmp C CXX)

set(CMAKE_C_STANDARD 11)
set(CMAKE_C_STANDARD_REQUIRED ON)
set(CMAKE_CXX_STANDARD 17)
set(CMAKE_CXX_STANDARD_REQUIRED ON)
set(CMAKE_POSITION_INDEPENDENT_CODE ON)

option(BUILD_JNI "Build the JNI shared library" OFF)
option(BUILD_WASM "Build the standalone WebAssembly module" OFF)

set(OPENH264_DIR "${CMAKE_CURRENT_SOURCE_DIR}/../openh264" CACHE PATH "OpenH264 checkout to build")
if(NOT EXISTS "${OPENH264_DIR}/codec/api/wels/codec_api.h")
  message(FATAL_ERROR "OpenH264 sources are missing at ${OPENH264_DIR}")
endif()

# OpenH264's version header is generated by its Makefile (and git-ignored).
# Reproduce it here so the decoder reports the release it was built from.
set(OPENH264_VERSION "2.7.0")
set(OPENH264_VERSION_DIR "${CMAKE_CURRENT_BINARY_DIR}/generated")
file(MAKE_DIRECTORY "${OPENH264_VERSION_DIR}")
file(WRITE "${OPENH264_VERSION_DIR}/version_gen.h"
  "#ifndef VERSION_GEN_H\n"
  "#define VERSION_GEN_H\n\n"
  "#define VERSION_NUMBER \"${OPENH264_VERSION}\"\n\n"
  "#endif // VERSION_GEN_H\n"
)

  ${OK_COMMON_SOURCES}
  ${OK_DECODER_SOURCES}

)

add_library(openh264-static STATIC ${OK_SOURCES})
target_include_directories(openh264-static
  PUBLIC "${CMAKE_CURRENT_SOURCE_DIR}/include"
  PRIVATE
    "${OPENH264_DIR}/codec/api/wels"
    "${OPENH264_DIR}/codec/common/inc"
    "${OPENH264_DIR}/codec/decoder/core/inc"
    "${OPENH264_DIR}/codec/decoder/plus/inc"
    "${OPENH264_VERSION_DIR}"
)
target_compile_definitions(openh264-static PRIVATE GENERATED_VERSION_HEADER)
if(CMAKE_CXX_COMPILER_ID MATCHES "GNU|Clang")
  target_compile_options(openh264-static PRIVATE
    -Wno-unused-function
    -Wno-unused-variable
    -Wno-unused-but-set-variable
  )
endif()

# The archive name is what the cinterop `-staticLibrary` flag and the JNI link
# line expect: libopenh264_kmp.a on Unix/MinGW.
set_target_properties(openh264-static PROPERTIES OUTPUT_NAME openh264_kmp)

if(UNIX)
  find_package(Threads REQUIRED)
  target_link_libraries(openh264-static PUBLIC Threads::Threads)
endif()

if(BUILD_JNI)
  add_library(openh264_jni SHARED "${CMAKE_CURRENT_SOURCE_DIR}/openh264_jni.cpp")
  target_link_libraries(openh264_jni PRIVATE openh264-static)
  set_target_properties(openh264_jni PROPERTIES OUTPUT_NAME openh264_jni)

  if(WIN32)
    # MinGW names shared libraries lib*.dll by default; the Kotlin loader uses
    # the unprefixed openh264_jni.dll. Only the MinGW runtime is linked
    # statically.
    set_target_properties(openh264_jni PROPERTIES PREFIX "")
    target_link_options(openh264_jni PRIVATE -static -static-libgcc -static-libstdc++)
  endif()

  if(ANDROID)
    find_library(log-lib log)
    if(log-lib)
      target_link_libraries(openh264_jni PRIVATE ${log-lib})
    endif()
    # 16 KB page-size devices (Android 15+).
    target_link_options(openh264_jni PRIVATE
      "-Wl,-z,max-page-size=16384"
      "-Wl,-z,common-page-size=16384"
    )
  else()
    # Derive the headers from JAVA_HOME first: FindJNI refuses to configure
    # when the target platform differs from the host (cross builds for MinGW).
    if(NOT JNI_INCLUDE_DIR AND DEFINED ENV{JAVA_HOME})
      set(JNI_INCLUDE_DIR "$ENV{JAVA_HOME}/include")
      if(WIN32)
        set(JNI_INCLUDE_DIR_PLATFORM "$ENV{JAVA_HOME}/include/win32")
      elseif(APPLE)
        set(JNI_INCLUDE_DIR_PLATFORM "$ENV{JAVA_HOME}/include/darwin")
      else()
        set(JNI_INCLUDE_DIR_PLATFORM "$ENV{JAVA_HOME}/include/linux")
      endif()
    endif()
    if(NOT JNI_INCLUDE_DIR)
      find_package(JNI REQUIRED)
    endif()
    if(JNI_INCLUDE_DIR)
      target_include_directories(openh264_jni PRIVATE "${JNI_INCLUDE_DIR}")
    endif()
    if(JNI_INCLUDE_DIR_PLATFORM)
      target_include_directories(openh264_jni PRIVATE "${JNI_INCLUDE_DIR_PLATFORM}")
    endif()
    if(JNI_INCLUDE_DIRS)
      target_include_directories(openh264_jni PRIVATE ${JNI_INCLUDE_DIRS})
    endif()
  endif()
endif()

if(BUILD_WASM OR EMSCRIPTEN OR CMAKE_SYSTEM_NAME STREQUAL "Emscripten")
  add_executable(openh264_wasm
    "${CMAKE_CURRENT_SOURCE_DIR}/openh264_wasm_entry.c"
    "${CMAKE_CURRENT_SOURCE_DIR}/openh264_wasm_compat.c"
  )
  target_link_libraries(openh264_wasm PRIVATE openh264-static)
  set_target_properties(openh264_wasm PROPERTIES SUFFIX ".wasm")
  target_link_options(openh264_wasm PRIVATE
    "-sSTANDALONE_WASM=1"
    "-sALLOW_MEMORY_GROWTH=1"
    "-sINITIAL_MEMORY=67108864"
    # The decoder's per-macroblock stack frames need more than the 64 KiB
    # Emscripten default for a 1080p picture.
    "-sSTACK_SIZE=8388608"
    "-sERROR_ON_UNDEFINED_SYMBOLS=1"
    "-sEXPORTED_FUNCTIONS=_ok_version_packed,_ok_version_string,_ok_api_build,_ok_decoder_create,_ok_decoder_destroy,_ok_decoder_decode,_ok_decoder_flush,_ok_decoder_frame_ready,_ok_decoder_frame_width,_ok_decoder_frame_height,_ok_decoder_frame_chroma_format,_ok_decoder_frame_pts,_ok_decoder_frame_plane,_ok_decoder_frame_blob,_ok_alloc,_ok_free,_malloc,_free"
    "--no-entry"
  )
endif()
